■ 한자 자동 수정:
python3 -c "
import re
path = '대상파일경로'
with open(path, encoding='utf-8') as f:
    c = f.read()
fixes = {'\u6790':'\uc11d','\u7d4c':'\uacbd','\u8981':'\uc694','\u7406':'\ub9ac','\u7fa9':'\uc758','\u5fb3':'\ub355'}
count = 0
for bad, good in fixes.items():
    n = c.count(bad)
    c = c.replace(bad, good)
    count += n
with open(path, 'w', encoding='utf-8') as f:
    f.write(c)
print('한자 수정:', count, '개')
"
